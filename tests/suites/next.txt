# ZX Spectrum Next on System/Next 24.11 trimmed to what NextZXOS boots from (corpus
# next/sysnext-24.11-min: machines/next's ROMs, menu.def and keymap.bin, nextzxos/, sys/).
# NextZXOS's menu is up by frame 700; the board's keys go in there (pc: = its PS/2 keyboard).
#
# name          machine  how   skip  frames  file  options
next_boot       zxnext   boot  850   50            sd:next/sysnext-24.11-min
# the NMI button from the menu, and F9 let go at once: NextZXOS's NMI menu
next_nmi        zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:nmi
next_f9         zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F9
# F9 held over a second: no NMI, the same hashes as next_boot
next_f9_long    zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F9/70
# F10, DRIVE: NextZXOS turns it off (NR 06 b4), the same hashes as next_boot
next_f10        zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F10
next_f3_60hz    zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F3
next_f8_speed   zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F8
next_f4_reset   zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F4
# a digit held with F9 is a function key: F9+4 a soft reset, F9+1 a hard one (the firmware again)
next_f9_4       zxnext   boot  850   50            sd:next/sysnext-24.11-min keys:700:pc:F9/10,702:pc:4
next_f9_1       zxnext   boot  1150  50            sd:next/sysnext-24.11-min keys:700:pc:F9/10,702:pc:1
